<script>
let injected = false;
function startInject(){
  if(injected) return;
  injected = true;
  const btn = document.getElementById("injectBtn");
  btn.innerText = "注入中...";
  btn.disabled = true;
  const progress = document.getElementById("progress");
  let p = 0;
  const taskList = [
    "正在寻找游戏进程...",
    "读取内存模块...",
    "绕过基础校验...",
    "注入核心模块...",
    "加载Anarchy模组包...",
    "注入完成！"
  ];
  let idx = 0;
  addLog(`> ${taskList[idx]}`);
  const timer = setInterval(()=>{
    p += 12;
    progress.style.width = p+"%";
    idx++;
    if(idx < taskList.length){
      addLog(`> ${taskList[idx]}`);
    }
    if(p >= 100){
      clearInterval(timer);
      btn.innerText = "注入成功";
      btn.classList.add("running");
      addLog("> ===== 客户端就绪，可以开启模块 =====");
    }
  },350)
}

function toggle(btn){
    if(!injected){
      addLog("> 错误：请先注入客户端！");
      return;
    }
    // 只有 从OFF点到ON【开启模块】的时候才判定概率
    if(btn.innerText === "OFF"){
        btn.innerText="ON";
        btn.classList.add("on");
        const name = btn.parentElement.querySelector("span").innerText.split(" ")[1];
        addLog(`> 模块【${name}】已启用`);
        // ✅每次开启模块都会随机判定，没有一次性限制
        if(Math.random() < 0.15){
          showPop();
        }
    }else{
        btn.innerText="OFF";
        btn.classList.remove("on");
        addLog(`> 模块关闭`);
    }
}
function addLog(text){
    const logDom = document.getElementById("log");
    logDom.innerHTML += text + "<br>";
    logDom.scrollTop = logDom.scrollHeight;
}
function showPop(){
  const pop = document.getElementById("alertPop");
  pop.style.display = "block";
  document.getElementById("mainPanel").classList.add("shake","warning-flash");
}
function closePop(){
  const pop = document.getElementById("alertPop");
  pop.style.display = "none";
  document.getElementById("mainPanel").classList.remove("shake","warning-flash");
  addLog("> ❗反作弊拦截，所有模块强制关闭，客户端锁定");
  // 全部模块强制切OFF
  document.querySelectorAll(".moduleBtn").forEach(b=>{
    b.innerText="OFF";
    b.classList.remove("on");
    b.disabled = true; // 模块按钮禁用
  })
  document.getElementById("injectBtn").disabled = true; //注入按钮禁用
}
// 重置全部，恢复所有功能
function resetAll(){
  injected = false;
  const injectBtn = document.getElementById("injectBtn");
  injectBtn.innerText = "开始注入";
  injectBtn.disabled = false;
  injectBtn.classList.remove("running");
  document.getElementById("progress").style.width = "0%";
  // 全部模块切回OFF，解除禁用
  document.querySelectorAll(".moduleBtn").forEach(b=>{
    b.innerText="OFF";
    b.classList.remove("on");
    b.disabled = false;
  })
  // 清空日志
  const logDom = document.getElementById("log");
  logDom.innerHTML = "> 客户端加载完成<br>> 等待点击【开始注入】";
  // 关闭弹窗，清除动画
  document.getElementById("alertPop").style.display="none";
  document.getElementById("mainPanel").classList.remove("shake","warning-flash");
}
</script>
